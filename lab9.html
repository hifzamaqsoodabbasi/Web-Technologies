// Example higher-order function from this experiment:
function performWithEffect(actionCallback, effectType) {
  // Apply visual effect
  applyVisualEffect(effectType);
  // Execute the callback function passed as argument
  actionCallback();
  // Log the action
  logAction(`Executed with ${effectType} effect`);
}
